"""Authenticate pre-provider local release evidence; no provider I/O."""
import hashlib
import json
import zipfile
from datetime import UTC, datetime
from pathlib import Path

from dmf_pulse.assurance.canonical import canonical_sha256, pretty_json
from dmf_pulse.assurance.manifests import RepositoryManifest, validate_repository_manifest
from dmf_pulse.evaluation.p01_a03_authority import REVIEW_PATHS, A03IndependentReview, load_a03_authority
from dmf_pulse.ingestion.openfootball.team_strength_data import authenticate

w=Path(__file__).resolve().parents[1]
e=w/'evidence/tickets/CURRENT-TEAM-STRENGTH-001V-P01-FALLBACK-A03'
public=w.parents[1]/'artifacts/public-prospective'
def digest(p):return hashlib.sha256(p.read_bytes()).hexdigest()
review=authenticate(A03IndependentReview.model_validate_json((e/'INDEPENDENT-REVIEW.json').read_bytes()))
assert tuple(row.path for row in review.files)==REVIEW_PATHS
for row in review.files:assert digest(w/row.path)==row.byte_sha256,row.path
inventory=json.loads((e/'HISTORICAL-INVENTORY.json').read_bytes())
sha=inventory.pop('semantic_sha256');assert canonical_sha256(inventory)==sha
count=0
for folder,items in inventory['original_artifacts'].items():
 for relative,expected in items.items():
  assert digest(public/folder/relative)==expected,(folder,relative)
  count+=1
for path,expected in inventory['protected_files'].items():assert digest(w/path)==expected,path
assert count==49
commands={}
for path in e.glob('COMMAND-*.json'):
 record=json.loads(path.read_bytes())
 if path.name=='COMMAND-SECRETS.json':
  assert record['exit_code']==1 and json.loads((e/'COMMAND-SECRETS-R02.json').read_bytes())['exit_code']==0
 else:assert record['exit_code']==0,path.name
 assert digest(e/record['output'])==record['output_byte_sha256'],path.name
 commands[path.name]=digest(path)
required={'FOCUSED-OFFLINE','FORMAT','LINT','MYPY','BUILD','FROZEN-SYNC','A02-D1-A03-INSTALLED-WHEEL','LEGACY-U-INSTALLED-WHEEL','IMMUTABILITY','A03-GITHUB-READ-ONLY','REPOSITORY','SECRETS','DIFF-CHECK'}
assert {'COMMAND-'+name+'.json' for name in required}<=set(commands)
network=json.loads((w/'review_pack/OFFLINE-TEST-NETWORK-RECEIPT.json').read_bytes())
assert network['pytest_exit_code']==network['denied_network_attempts']==network['provider_network_calls']==0
(e/'OFFLINE-TEST-NETWORK-RECEIPT.json').write_bytes((w/'review_pack/OFFLINE-TEST-NETWORK-RECEIPT.json').read_bytes())
manifest=RepositoryManifest.model_validate_json((e/'current_manifest.json').read_bytes())
assert not validate_repository_manifest(w,manifest)
wheel=w/'dist/dmf_pulse-0.2.0-py3-none-any.whl'
with zipfile.ZipFile(wheel) as z:
 for name in REVIEW_PATHS:
  if name.startswith('src/dmf_pulse/'):assert z.read(name.removeprefix('src/'))==(w/name).read_bytes(),name
 assert z.read('dmf_pulse/evaluation/resources/evaluation/team_strength_p01_fallback_a03.json')==(w/'config/evaluation/team_strength_p01_fallback_a03.json').read_bytes()
v={'schema_version':'p01-a03-final-local-audit-v1','audited_at':datetime.now(UTC).isoformat(),'immutable_parent':'f96eaa3e7261d0db6f23bc8eb7dee18917272d00','authority_sha256':load_a03_authority().semantic_sha256,'review_sha256':review.semantic_sha256,'reviewed_files_authenticated':len(review.files),'historical_artifacts_unchanged':count,'protected_files_unchanged':len(inventory['protected_files']),'current_manifest_files':len(manifest.files),'command_record_byte_hashes':commands,'offline_network_receipt':network,'wheel_byte_sha256':digest(wheel),'sdist_byte_sha256':digest(w/'dist/dmf_pulse-0.2.0.tar.gz'),'provider_requests':0,'p01_scored':False,'credentials_inspected':0,'production_active':False,'release_state':'READY_FOR_COMMIT_AND_EXACT_SHA_CI','live_execution_permitted_before_ci':False,'resolved_checkpoint_failures':[{'command':'COMMAND-SECRETS.json','resolved_by':'COMMAND-SECRETS-R02.json','reason':'Governance authorization status renamed human_decision_status to avoid credential-key ambiguity; scanner unchanged, authority and independent review resealed.'}]}
v['semantic_sha256']=canonical_sha256(v)
with (e/'FINAL-LOCAL-AUDIT.json').open('x',encoding='utf-8',newline='\n') as stream:stream.write(pretty_json(v))
print(pretty_json(v),end='')
