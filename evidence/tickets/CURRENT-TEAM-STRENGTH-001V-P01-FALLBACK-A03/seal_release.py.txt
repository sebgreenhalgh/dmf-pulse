def digest(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def run(args):return subprocess.run(args,cwd=w,capture_output=True,check=True,timeout=60).stdout
def persist(category,name,payload):
 path=root/category/name;path.parent.mkdir(parents=True,exist_ok=True)
 with path.open('xb') as stream:stream.write(payload);stream.flush();os.fsync(stream.fileno())
 assert path.read_bytes()==payload
 return path

assert remote==sha

assert binding.file_review_sha256==review.semantic_sha256
for item in review.files:assert digest(w/item.path)==item.byte_sha256,item.path

local=json.loads((e/'FINAL-LOCAL-AUDIT.json').read_bytes());s=local.pop('semantic_sha256');assert canonical_sha256(local)==s

control.verify(binding,expected_review_bytes=(e/'INDEPENDENT-REVIEW.json').read_bytes())
