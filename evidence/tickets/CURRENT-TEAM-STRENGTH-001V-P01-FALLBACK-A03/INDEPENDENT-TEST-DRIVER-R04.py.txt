"""Independent review execution: deny networking and preserve exact test output."""

import hashlib
import json
import socket
import sys
import uuid
from datetime import UTC, datetime
from pathlib import Path

attempts = []


def denied(*args, **kwargs):
    attempts.append("NETWORK_REFUSED")
    raise AssertionError("Independent A03 review is strictly offline")


socket.create_connection = socket.getaddrinfo = socket.socket.connect = denied
socket.socket.connect_ex = socket.socket.sendto = denied

import pytest

start = datetime.now(UTC)
arguments = [
    "tests/unit/evaluation/test_p01_fallback_a03.py",
    "tests/unit/evaluation/test_p01_fallback_d1.py",
    "-q",
    "-p",
    "no:cacheprovider",
    "--basetemp",
    "review_pack/independent-a03-pytest-" + uuid.uuid4().hex,
    "--tb=short",
]
code = int(pytest.main(arguments))
record = {
    "arguments": arguments,
    "started_at": start.isoformat(),
    "completed_at": datetime.now(UTC).isoformat(),
    "exit_code": code,
    "denied_network_attempts": len(attempts),
    "provider_requests": 0,
    "credentials_inspected": 0,
    "real_p01_scoring_performed": False,
}
Path("review_pack/INDEPENDENT-A03-TEST-COMMAND-R04.json").write_text(
    json.dumps(record, indent=2) + "\n", encoding="utf-8", newline="\n"
)
print(json.dumps(record, sort_keys=True))
sys.exit(code if not attempts else 1)
