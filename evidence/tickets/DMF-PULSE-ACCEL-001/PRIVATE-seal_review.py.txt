def read(path):
    return json.loads(path.read_bytes())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode()


def numerical_payload(value):
    if isinstance(value, dict):
        return {key: numerical_payload(item) for key, item in value.items() if "sha256" not in key}
    if isinstance(value, list):
        return [numerical_payload(item) for item in value]
    if isinstance(value, str):
        return re.sub(r"screen SHA256: [0-9a-f]{64}", "screen SHA256: BUILD_BOUND_IDENTITY", value)
    return value

    changes = read(baseline_root / f"decision-diff-{horizon}.json")

        path, left, right = change["path"], change["before"], change["after"]
