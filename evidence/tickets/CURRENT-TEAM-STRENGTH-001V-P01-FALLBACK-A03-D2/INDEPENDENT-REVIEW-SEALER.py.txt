def digest(value):
    return hashlib.sha256(value).hexdigest()


def canonical(value):
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode()


def sealed(path):
    value = json.loads(path.read_bytes())
    body = dict(value)
    expected = body.pop("semantic_sha256")
    assert digest(canonical(body)) == expected, str(path)
    return value


def write(name, value):
    value["semantic_sha256"] = digest(canonical(value))
    with (OUT / name).open("x", encoding="utf-8", newline="\n") as handle:
        handle.write(json.dumps(value, sort_keys=True, ensure_ascii=False, indent=2) + "\n")
    return value["semantic_sha256"]
