import json
import socket
import sys
from pathlib import Path

attempts = []
def blocked(*a, **kw):
    attempts.append('NETWORK_REFUSED')
    raise AssertionError('D2 release tests must remain offline')
socket.create_connection = socket.getaddrinfo = socket.socket.connect = blocked
socket.socket.connect_ex = socket.socket.sendto = blocked
import pytest
code = pytest.main(sys.argv[1:])
record = {'provider_network_calls':0,'denied_network_attempts':len(attempts),
    'openligadb_requests':0,'openfootball_live_requests':0,'fpl_requests':0,'odds_requests':0,
    'credentials_inspected':0,'real_p01_scoring_performed':False,'pytest_exit_code':int(code)}
Path('review_pack/OFFLINE-TEST-NETWORK-RECEIPT.json').write_text(json.dumps(record,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps(record,sort_keys=True))
sys.exit(int(code) if not attempts else 1)
