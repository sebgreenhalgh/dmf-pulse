w=Path(__file__).resolve().parents[1]

def digest(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def run(args):return subprocess.run(args,cwd=w,capture_output=True,check=True,timeout=60).stdout
def persist(category,payload,extension='.json'):
    key=hashlib.sha256(payload).hexdigest()
    path=root/category/(key+extension);path.parent.mkdir(parents=True,exist_ok=True)
    with path.open('xb') as f:f.write(payload);f.flush();os.fsync(f.fileno())
    assert digest(path)==key
    return path,key
def sealed(path):
    value=json.loads(path.read_bytes());key=value.pop('semantic_sha256')
    assert canonical_sha256(value)==key,path
    return value,key
